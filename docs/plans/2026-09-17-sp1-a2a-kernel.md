# V-SI SP1 A2A 커널 Implementation Plan

> 이 계획은 태스크 단위로 실행하도록 쓰였다. 각 단계는 체크박스(`- [ ]`) 로 추적한다.

**Goal:** LLM을 한 번도 호출하지 않고 기획→개발→QA/보안→개발 환류 전 구간이 동작하고, 단일 트레이스로 관측되며, 오케스트레이터를 SIGKILL 해도 복구되는 A2A 커널을 만든다.

**Architecture:** 허브-스포크. 오케스트레이터가 A2A HTTP+JSON으로 스텁 에이전트 4종을 호출하고, 워크플로 상태 전이와 도메인 이벤트를 한 트랜잭션에 기록한다(트랜잭셔널 아웃박스). Event Gateway가 아웃박스를 테일해 WebSocket으로 React 그래프 UI에 팬아웃하고, 전 서비스가 OTel Collector를 거쳐 Jaeger로 트레이스를 보낸다. A2A Task 상태(SDK 소관)와 워크플로 상태(우리 소관)를 두 층으로 분리한다.

**Tech Stack:** Python 3.11 · FastAPI · `a2a-sdk` 1.1.2 · SQLAlchemy 2.x (async) · PostgreSQL 16 · pytest + pytest-asyncio · OpenTelemetry · React 18 + Vite · Docker Compose

**Spec:** `docs/specs/2026-09-17-a2a-kernel-design.md`

## Global Constraints

- Python **3.11 이상**. `a2a-sdk==1.1.2` 로 핀 고정.
- A2A 전송은 **HTTP+JSON 바인딩만** 사용한다. gRPC·SSE 스트리밍은 범위 밖.
- 에이전트는 **LLM을 호출하지 않는다.** SP1 전 과정의 토큰 비용은 0원이어야 한다.
- 에이전트는 다른 에이전트를 직접 호출하지 않는다. **모든 통신은 오케스트레이터 경유.**
- **재시도 권한은 오케스트레이터만** 갖는다.
- Task는 **불변**이다. 재시도·환류는 항상 새 행을 만들고 `revision_of`로 계보를 잇는다.
- `input-required`는 **사람 승인 대기 전용**이다. 재작업에 쓰지 않는다.
- 타임아웃은 **안쪽 < 바깥쪽** 부등식을 만족해야 하며, 위반 시 기동을 실패시킨다.
- `verdict`는 LLM 주장이 아니라 **도구 종료 코드**에서 도출한다.
- SDK 테이블은 스키마 `a2a`, 오케스트레이터 테이블은 스키마 `public`에 둔다.
- 푸시 토큰·시크릿은 **AgentCard에 절대 포함하지 않는다.**
- 모든 커밋 메시지는 한국어 본문, Conventional Commits 접두사.

```python
# packages/orchestrator/policy.py
from __future__ import annotations
from dataclasses import dataclass
from collections.abc import Mapping


class TimeoutConfigError(ValueError):
    """타임아웃 계층이 안쪽 < 바깥쪽 부등식을 위반했다."""


@dataclass(frozen=True)
class TimeoutConfig:
    tool_s: int
    executor_s: int
    step_s: int
    run_s: int

    def validate(self) -> None:
        layers = [("tool_s", self.tool_s), ("executor_s", self.executor_s),
                  ("step_s", self.step_s), ("run_s", self.run_s)]
        for (inner_name, inner), (outer_name, outer) in zip(layers, layers[1:]):
            if inner >= outer:
                raise TimeoutConfigError(
                    f"{outer_name}({outer})는 {inner_name}({inner})보다 커야 한다"
                )

    @classmethod
    def from_env(cls, env: Mapping[str, str]) -> "TimeoutConfig":
        cfg = cls(
            tool_s=int(env.get("VSI_TIMEOUT_TOOL_S", 30)),
            executor_s=int(env.get("VSI_TIMEOUT_EXECUTOR_S", 300)),
            step_s=int(env.get("VSI_TIMEOUT_STEP_S", 900)),
            run_s=int(env.get("VSI_TIMEOUT_RUN_S", 3600)),
        )
        cfg.validate()
        return cfg
```

```python
# packages/orchestrator/workflow.py
from __future__ import annotations
from enum import StrEnum


class RequirementState(StrEnum):
    PLANNED = "planned"
    IMPLEMENTING = "implementing"
    VERIFYING = "verifying"
    REMEDIATING = "remediating"
    BLOCKED = "blocked"
    ACCEPTED = "accepted"
    ESCALATED = "escalated"


class WorkflowSignal(StrEnum):
    PLAN_READY = "plan_ready"
    DEV_DONE = "dev_done"
    VERDICTS_PASS = "verdicts_pass"
    VERDICTS_FAIL = "verdicts_fail"
    APPROVAL_REQUIRED = "approval_required"
    APPROVAL_GRANTED = "approval_granted"
    LIMIT_EXCEEDED = "limit_exceeded"


class IllegalTransition(RuntimeError):
    """현재 상태에서 허용되지 않는 신호."""


S, Sig = RequirementState, WorkflowSignal

TERMINAL: frozenset[RequirementState] = frozenset({S.ACCEPTED, S.ESCALATED})

_TABLE: dict[tuple[RequirementState, WorkflowSignal], RequirementState] = {
    (S.PLANNED, Sig.PLAN_READY): S.IMPLEMENTING,
    (S.IMPLEMENTING, Sig.DEV_DONE): S.VERIFYING,
    (S.IMPLEMENTING, Sig.APPROVAL_REQUIRED): S.BLOCKED,
    (S.BLOCKED, Sig.APPROVAL_GRANTED): S.IMPLEMENTING,
    (S.VERIFYING, Sig.VERDICTS_PASS): S.ACCEPTED,
    (S.VERIFYING, Sig.VERDICTS_FAIL): S.REMEDIATING,
    (S.REMEDIATING, Sig.DEV_DONE): S.IMPLEMENTING,
    (S.REMEDIATING, Sig.LIMIT_EXCEEDED): S.ESCALATED,
}


def next_state(current: RequirementState, signal: WorkflowSignal) -> RequirementState:
    try:
        return _TABLE[(current, signal)]
    except KeyError:
        raise IllegalTransition(f"{current.name} 상태에서 {signal.name} 신호는 허용되지 않는다") from None
```

```python
# packages/orchestrator/idempotency.py
from __future__ import annotations
import hashlib
from collections.abc import Sequence


def idempotency_key(requirement_id: str, agent: str, revision: int,
                    input_hashes: Sequence[str]) -> str:
    """같은 (요구사항, 에이전트, 환류 회차, 입력)이면 같은 키.

    입력 Artifact 순서는 의미가 없으므로 정렬해 안정화한다.
    revision이 포함되므로 환류로 인한 정당한 재실행은 다른 키가 된다.
    """
    material = "|".join([requirement_id, agent, str(revision), *sorted(input_hashes)])
    return hashlib.sha256(material.encode("utf-8")).hexdigest()
```

```python
import pytest
from sqlalchemy import select
from orchestrator.models import WorkflowRequirement, OutboxEvent
from orchestrator.outbox import record_event
from orchestrator.workflow import RequirementState


async def test_state_change_and_event_commit_together(session):
    req = WorkflowRequirement(requirement_id="REQ-001", title="회원가입",
                              state=RequirementState.PLANNED, run_id="run-1")
    session.add(req)
    await session.commit()

    req.state = RequirementState.IMPLEMENTING
    record_event(session, "requirement", "REQ-001", "state_changed",
                 {"to": "implementing"})
    await session.commit()

    events = (await session.execute(select(OutboxEvent))).scalars().all()
    assert len(events) == 1
    assert events[0].published_at is None
    assert events[0].payload["to"] == "implementing"


async def test_rollback_discards_both(session):
    req = WorkflowRequirement(requirement_id="REQ-002", title="로그인",
                              state=RequirementState.PLANNED, run_id="run-1")
    session.add(req)
    await session.commit()

    req.state = RequirementState.IMPLEMENTING
    record_event(session, "requirement", "REQ-002", "state_changed", {"to": "implementing"})
    await session.rollback()
```

```python
# packages/orchestrator/models.py
from __future__ import annotations
import uuid
from datetime import datetime
from sqlalchemy import (BigInteger, DateTime, ForeignKey, Index, Integer, String,
                        UniqueConstraint, func, text)
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column


class Base(DeclarativeBase):
    """오케스트레이터 소유 스키마(public). SDK 테이블은 스키마 a2a에 별도."""


def _uuid() -> str:
    return str(uuid.uuid4())


class WorkflowRequirement(Base):
    __tablename__ = "workflow_requirements"
    requirement_id: Mapped[str] = mapped_column(String, primary_key=True)
    title: Mapped[str] = mapped_column(String, nullable=False)
    state: Mapped[str] = mapped_column(String, nullable=False)
    revision: Mapped[int] = mapped_column(Integer, nullable=False, default=1)
    max_revisions: Mapped[int] = mapped_column(Integer, nullable=False, default=3)
    run_id: Mapped[str] = mapped_column(String, nullable=False)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=func.now())
    updated_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now())


class WorkflowTask(Base):
    __tablename__ = "workflow_tasks"
    task_id: Mapped[str] = mapped_column(String, primary_key=True, default=_uuid)
    requirement_id: Mapped[str] = mapped_column(ForeignKey("workflow_requirements.requirement_id"))
    agent: Mapped[str] = mapped_column(String, nullable=False)
```

```python
# packages/stub_agent/scenario.py
from __future__ import annotations
from dataclasses import dataclass, field
import yaml


class DuplicateCallError(RuntimeError):
    """같은 멱등성 키로 두 번 호출됐다. 오케스트레이터의 멱등성 위반."""


@dataclass
class AgentScenario:
    agent: str
    verdicts: list[str] = field(default_factory=list)
    failures: dict[int, str] = field(default_factory=dict)
    latency_ms: int = 0
    _cursor: int = 0
    _seen_keys: dict[str, int] = field(default_factory=dict)

    @classmethod
    def from_yaml(cls, path: str, agent: str) -> "AgentScenario":
        with open(path, encoding="utf-8") as fh:
            doc = yaml.safe_load(fh)
        spec = doc.get("agents", {}).get(agent, {}) or {}
        failures = {
            int(k.removeprefix("attempt_")): v
            for k, v in spec.items() if k.startswith("attempt_")
        }
        return cls(agent=agent, verdicts=list(spec.get("verdicts", [])),
                   failures=failures, latency_ms=int(spec.get("latency_ms", 0)))

    def next_verdict(self) -> str | None:
        if not self.verdicts:
            return None
        idx = min(self._cursor, len(self.verdicts) - 1)
        self._cursor += 1
        return self.verdicts[idx]

    def failure_for_attempt(self, n: int) -> str | None:
        return self.failures.get(n)

    def record_call(self, idempotency_key: str) -> int:
        if idempotency_key in self._seen_keys:
            raise DuplicateCallError(f"중복 호출: {idempotency_key}")
        self._seen_keys[idempotency_key] = 1
        return 1
```

```python
from __future__ import annotations
import time
from stub_agent.scenario import AgentScenario

ARTIFACT_KIND: dict[str, str] = {
    "planner": "requirements",
    "dev": "source_code",
    "qa": "test_report",
    "security": "security_report",
}


class StubCrash(RuntimeError):
    """시나리오가 주입한 크래시."""


class StubExecutor:
    """LLM을 호출하지 않는다. 시나리오가 지시한 결과만 만든다."""

    def __init__(self, scenario: AgentScenario, agent: str) -> None:
        self.scenario = scenario
        self.agent = agent

    def build_payload(self, attempt: int) -> dict:
        if self.scenario.failure_for_attempt(attempt) == "crash":
            raise StubCrash(f"{self.agent} attempt {attempt} 크래시 주입")
        if self.scenario.latency_ms:
            time.sleep(self.scenario.latency_ms / 1000)

        verdict = self.scenario.next_verdict()
        payload: dict = {"kind": ARTIFACT_KIND[self.agent], "agent": self.agent}
        if verdict is not None:
            # verdict는 주장이 아니라 종료 코드에서 도출한다.
            payload["exit_code"] = 0 if verdict == "PASS" else 1
            payload["verdict"] = "PASS" if payload["exit_code"] == 0 else "FAIL"
        else:
            payload["exit_code"] = 0
        return payload

    async def execute(self, context, event_queue) -> None:
        """SDK AgentExecutor 인터페이스. 정확한 시그니처는
        inspect.signature(a2a.server.agent_execution.AgentExecutor.execute)로 확인해 맞춘다."""
        payload = self.build_payload(attempt=1)
        await event_queue.enqueue_event(payload)
```

```python
# packages/orchestrator/push_receiver.py
from __future__ import annotations
import hmac
from collections.abc import Awaitable, Callable
from fastapi import APIRouter, Header, HTTPException, Request, status

PUSH_TOKEN_HEADER = "X-A2A-Notification-Token"


def create_push_router(token: str,
                       on_notification: Callable[[dict], Awaitable[None]]) -> APIRouter:
    """SDK BasePushNotificationSender가 보내는 콜백을 받는다.

    토큰은 등록 시 우리가 발급해 push notification config에 저장된 값이며,
    AgentCard에는 포함되지 않는다.
    """
    router = APIRouter()

    @router.post("/push", status_code=status.HTTP_202_ACCEPTED)
    async def receive(request: Request,
                      x_a2a_notification_token: str | None = Header(default=None)) -> dict:
        if x_a2a_notification_token is None or not hmac.compare_digest(
            x_a2a_notification_token, token
        ):
            raise HTTPException(status_code=401, detail="invalid notification token")
        await on_notification(await request.json())
        return {"accepted": True}

    return router
```

```python
# packages/orchestrator/engine.py
from __future__ import annotations
from datetime import datetime, timezone
from orchestrator.idempotency import idempotency_key
from orchestrator.models import Artifact, WorkflowRequirement, WorkflowTask
from orchestrator.outbox import record_event
from orchestrator.policy import TimeoutConfig
from orchestrator.workflow import RequirementState, WorkflowSignal, next_state

PIPELINE = ["planner", "dev"]
VERIFIERS = ["qa", "security"]


class WorkflowEngine:
    def __init__(self, session_maker, clients: dict, timeouts: TimeoutConfig) -> None:
        self._sm = session_maker
        self._clients = clients
        self._timeouts = timeouts

    async def start(self, requirement_id: str, title: str, run_id: str) -> None:
        async with self._sm() as s:
            s.add(WorkflowRequirement(requirement_id=requirement_id, title=title,
                                      state=RequirementState.PLANNED, run_id=run_id))
            record_event(s, "requirement", requirement_id, "state_changed",
                         {"to": RequirementState.PLANNED})
            await s.commit()
        await self._dispatch(requirement_id, "planner")

    async def _dispatch(self, requirement_id: str, agent: str) -> None:
        async with self._sm() as s:
            req = await s.get(WorkflowRequirement, requirement_id)
            key = idempotency_key(requirement_id, agent, req.revision, [])
            task = WorkflowTask(requirement_id=requirement_id, agent=agent,
                                revision=req.revision, idempotency_key=key,
                                state="submitted")
            s.add(task)
            record_event(s, "task", task.task_id, "task_submitted",
                         {"agent": agent, "requirement_id": requirement_id,
                          "revision": req.revision})
            await s.commit()
            task_id = task.task_id
        a2a_id = await self._clients[agent].submit({"requirement_id": requirement_id}, key)
        async with self._sm() as s:
            t = await s.get(WorkflowTask, task_id)
            t.a2a_task_id, t.state = a2a_id, "working"
            await s.commit()

    async def on_task_completed(self, a2a_task_id: str, payload: dict) -> None:
        """푸시 콜백 진입점. 완료된 Task를 기록하고 다음 단계를 결정한다."""
        async with self._sm() as s:
            task = (await s.execute(
                WorkflowTask.__table__.select().where(
                    WorkflowTask.__table__.c.a2a_task_id == a2a_task_id)
            )).first()
            assert task is not None, f"알 수 없는 a2a_task_id: {a2a_task_id}"
            t = await s.get(WorkflowTask, task.task_id)
            t.state = "completed"
            t.completed_at = datetime.now(timezone.utc)
            if "verdict" in payload:
                t.verdict = "PASS" if payload["exit_code"] == 0 else "FAIL"
            s.add(Artifact(requirement_id=t.requirement_id, producer_task=t.task_id,
                           kind=payload["kind"], version=t.revision,
                           content=payload, sha256=str(hash(str(payload)))))
            record_event(s, "task", t.task_id, "task_completed",
                         {"agent": t.agent, "verdict": t.verdict})
            await s.commit()
            requirement_id, agent = t.requirement_id, t.agent
        await self._advance(requirement_id, agent)

    async def _advance(self, requirement_id: str, finished_agent: str) -> None:
        if finished_agent == "planner":
            await self._transition(requirement_id, WorkflowSignal.PLAN_READY)
            await self._dispatch(requirement_id, "dev")
        elif finished_agent == "dev":
            await self._transition(requirement_id, WorkflowSignal.DEV_DONE)
            for verifier in VERIFIERS:
                await self._dispatch(requirement_id, verifier)
        else:
            await self._maybe_finish(requirement_id)

    async def _transition(self, requirement_id: str, signal: WorkflowSignal) -> None:
        async with self._sm() as s:
            req = await s.get(WorkflowRequirement, requirement_id)
            req.state = next_state(RequirementState(req.state), signal)
            record_event(s, "requirement", requirement_id, "state_changed",
                         {"to": req.state, "signal": signal})
            await s.commit()

    async def _maybe_finish(self, requirement_id: str) -> None:
        """qa/security 두 verdict가 모두 도착했을 때만 판정한다."""
        async with self._sm() as s:
            req = await s.get(WorkflowRequirement, requirement_id)
            rows = (await s.execute(
                WorkflowTask.__table__.select().where(
                    (WorkflowTask.__table__.c.requirement_id == requirement_id)
                    & (WorkflowTask.__table__.c.revision == req.revision)
                    & (WorkflowTask.__table__.c.agent.in_(VERIFIERS)))
            )).fetchall()
            verdicts = [r.verdict for r in rows if r.verdict is not None]
        if len(verdicts) < len(VERIFIERS):
            return
        if all(v == "PASS" for v in verdicts):
            await self._transition(requirement_id, WorkflowSignal.VERDICTS_PASS)
        else:
            await self._transition(requirement_id, WorkflowSignal.VERDICTS_FAIL)

    async def state_of(self, requirement_id: str) -> RequirementState:
        async with self._sm() as s:
            req = await s.get(WorkflowRequirement, requirement_id)
            return RequirementState(req.state)
```

```python
# packages/orchestrator/retry.py
from __future__ import annotations
from enum import StrEnum
import httpx


class FailureClass(StrEnum):
    TRANSPORT = "transport"
    EXECUTION = "execution"
    POISON = "poison"


_MAX_ATTEMPTS = {
    FailureClass.TRANSPORT: 3,
    FailureClass.EXECUTION: 2,
    FailureClass.POISON: 1,   # 재시도해도 같은 실패가 확정적이다
}


def classify(exc: Exception) -> FailureClass:
    if isinstance(exc, httpx.HTTPStatusError):
        return FailureClass.POISON if 400 <= exc.response.status_code < 500 else FailureClass.TRANSPORT
    if isinstance(exc, (httpx.ConnectError, httpx.TimeoutException, httpx.NetworkError)):
        return FailureClass.TRANSPORT
    if isinstance(exc, (ValueError, TypeError, KeyError)):
        return FailureClass.POISON
    return FailureClass.EXECUTION


def max_attempts(fc: FailureClass) -> int:
    return _MAX_ATTEMPTS[fc]


def backoff_seconds(attempt: int) -> float:
    return min(0.5 * (2 ** (attempt - 1)), 8.0)
```

```python
# packages/orchestrator/reconciler.py
from __future__ import annotations
import asyncio
from sqlalchemy import select
from orchestrator.models import WorkflowRequirement, WorkflowTask
from orchestrator.workflow import RequirementState

ACTIVE = (RequirementState.IMPLEMENTING, RequirementState.VERIFYING,
          RequirementState.REMEDIATING)


class Reconciler:
    """이벤트를 재생하지 않는다. 현재 상태를 관찰해 목표 상태로 수렴시킨다.

    크래시 복구가 특수 경로가 아니라 평시에도 도는 루프이므로 테스트가 쉽다.
    """

    def __init__(self, session_maker, engine, interval_s: float = 10.0) -> None:
        self._sm = session_maker
        self._engine = engine
        self._interval = interval_s

    async def reconcile_once(self) -> int:
        async with self._sm() as s:
            stuck = (await s.execute(
                select(WorkflowRequirement).where(
                    WorkflowRequirement.state.in_([str(x) for x in ACTIVE]))
            )).scalars().all()
            plan: list[tuple[str, str]] = []
            for req in stuck:
                rows = (await s.execute(
                    select(WorkflowTask).where(
                        (WorkflowTask.requirement_id == req.requirement_id)
                        & (WorkflowTask.revision == req.revision))
                )).scalars().all()
                open_tasks = [t for t in rows if t.state in ("submitted", "working")]
                if open_tasks:
                    continue          # 진행 중 — 손대지 않는다
                done = {t.agent for t in rows if t.state == "completed"}
                for agent in ("planner", "dev", "qa", "security"):
                    if agent not in done:
                        plan.append((req.requirement_id, agent))
                        break

        for requirement_id, agent in plan:
            await self._engine._dispatch(requirement_id, agent)
        return len(plan)

    async def run_forever(self) -> None:
        while True:
            try:
                await self.reconcile_once()
            except Exception:  # 루프는 절대 죽지 않는다
                pass
            await asyncio.sleep(self._interval)
```

백그라운드 태스크로 띄운다.

```python
from __future__ import annotations
import asyncio
import os
from datetime import datetime, timezone
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from sqlalchemy import select
from orchestrator.db import make_engine, session_factory
from orchestrator.models import OutboxEvent

POLL_S = 0.5
app = FastAPI(title="v-si event gateway")
_clients: set[WebSocket] = set()
_maker = session_factory(make_engine(os.environ["VSI_DATABASE_URL"]))


async def tail_outbox(session_maker, last_id: int, limit: int = 100) -> list[dict]:
    async with session_maker() as s:
        rows = (await s.execute(
            select(OutboxEvent).where(OutboxEvent.event_id > last_id)
            .order_by(OutboxEvent.event_id).limit(limit))).scalars().all()
        out = []
        for r in rows:
            out.append({"event_id": r.event_id, "aggregate": r.aggregate,
                        "aggregate_id": r.aggregate_id, "event_type": r.event_type,
                        "payload": r.payload})
            r.published_at = datetime.now(timezone.utc)
        await s.commit()
        return out


async def _pump() -> None:
    """마지막 published 이후부터 이어서 테일한다. 게이트웨이가 죽어도 유실이 없다."""
    last_id = 0
    while True:
        for event in await tail_outbox(_maker, last_id):
            last_id = event["event_id"]
            for ws in list(_clients):
                try:
                    await ws.send_json(event)
                except Exception:
                    _clients.discard(ws)
        await asyncio.sleep(POLL_S)


@app.on_event("startup")
async def _start() -> None:
    asyncio.create_task(_pump())
```

```ts
// web/src/useEventStream.ts
import { useEffect, useState } from "react";

export type VsiEvent = {
  event_id: number; aggregate: string; aggregate_id: string;
  event_type: string; payload: Record<string, unknown>;
};
export type StreamState = { events: VsiEvent[]; activeAgents: Set<string> };

export function reduceEvents(state: StreamState, e: VsiEvent): StreamState {
  const activeAgents = new Set(state.activeAgents);
  const agent = e.payload.agent as string | undefined;
  if (agent && e.event_type === "task_submitted") activeAgents.add(agent);
  if (agent && (e.event_type === "task_completed" || e.event_type === "task_failed"))
    activeAgents.delete(agent);
  return { events: [...state.events, e], activeAgents };
}

export function useEventStream(url: string): StreamState {
  const [state, setState] = useState<StreamState>({ events: [], activeAgents: new Set() });
  useEffect(() => {
    const ws = new WebSocket(url);
    ws.onmessage = (m) => setState((s) => reduceEvents(s, JSON.parse(m.data)));
    return () => ws.close();
  }, [url]);
  return state;
}
```
